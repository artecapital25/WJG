function generarOT() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const resumenSheet = ss.getSheetByName("RESUMEN_COT");
  const cotizacionesSheet = ss.getSheetByName("COTIZACIONES");
  const otSheet = ss.getSheetByName("OT");

  // Traer datos de RESUMEN_COT
  const resumenData = resumenSheet.getDataRange().getValues();
  let cotAceptadas = resumenData.filter((row, i) => i > 0 && row[1] == "Aceptada").map(row => row[0]);

  if (cotAceptadas.length === 0) {
    SpreadsheetApp.getUi().alert("No hay cotizaciones aceptadas.");
    return;
  }

  // Traer datos de COTIZACIONES
  const cotizacionesData = cotizacionesSheet.getDataRange().getValues();

  // Calcular número de OT
  let lastRow = otSheet.getLastRow();
  let numeroOT = lastRow > 1 ? parseInt(otSheet.getRange(lastRow, 1).getValue().toString().replace("OT-", "")) : 0;

  // Recorrer cotizaciones aceptadas y generar OT
  for (let i = 0; i < cotAceptadas.length; i++) {
    let idCot = cotAceptadas[i];
    let cliente = resumenData.find(r => r[0] === idCot)[2]; // Cliente desde RESUMEN_COT (col C)

    for (let j = 1; j < cotizacionesData.length; j++) {
      if (cotizacionesData[j][0] === idCot) { 
        numeroOT++;
        let item = cotizacionesData[j][1];       // Col B = Item
        let descripcion = cotizacionesData[j][2]; // Col C = Descripción
        let cantidad = cotizacionesData[j][3];    // Col D = Cantidad
        let unidad = cotizacionesData[j][4];      // Col E = Unidad

        // Insertar fila en OT
        otSheet.appendRow([
          "OT-" + ("000" + numeroOT).slice(-4), // # OT
          idCot,                                // ID Cotizacion
          cliente,                              // Cliente
          new Date(),                           // Fecha OT
          item,                                 // Item
          descripcion,                          // Descripcion
          cantidad,                             // Cantidad
          unidad,                               // Unidad
          "",                                   // Tiempo desarrollo
          "",                                   // Tiempo armado
          ""                                    // Tiempo pintura
        ]);
      }
    }
  }

  SpreadsheetApp.getUi().alert("Órdenes de trabajo generadas correctamente.");
}

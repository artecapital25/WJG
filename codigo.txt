function ajustarFilasDesdeBoton() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("FORMATO DE COT");
  const cantidad = parseInt(hoja.getRange("K2").getValue(), 10);

  if (isNaN(cantidad) || cantidad <= 0) {
    SpreadsheetApp.getUi().alert("La cantidad debe ser mayor a 0");
    return;
  }

  const filaInicio = 10;
  const columnas = 6;

  const datos = hoja.getRange(filaInicio, 1, hoja.getMaxRows() - filaInicio + 1, 1).getValues();
  let filaFija = hoja.getMaxRows();

  for (let i = 0; i < datos.length; i++) {
    const valor = (datos[i][0] || "").toString().toUpperCase().trim();
    if (valor === "NOTAS" || valor === "TOTAL") {
      filaFija = filaInicio + i;
      break;
    }
  }

  const filasExistentes = Math.max(0, filaFija - filaInicio);
  const diferencia = cantidad - filasExistentes;

  if (filasExistentes > 0) {
    hoja.getRange(filaInicio, 1, filasExistentes, columnas).breakApart();
  }

  if (diferencia > 0) {
    hoja.insertRowsBefore(filaFija, diferencia);
  } else if (diferencia < 0) {
    hoja.deleteRows(filaInicio + cantidad, Math.abs(diferencia));
  }

  hoja.getRange(filaInicio, 1, cantidad, hoja.getLastColumn()).clearContent();

  hoja.getDrawings().forEach(dibujo => {
    const fila = dibujo.getContainerInfo().getAnchorRow();
    if (fila >= filaInicio && fila < filaInicio + cantidad) {
      dibujo.remove();
    }
  });

  hoja.setColumnWidth(4, 200);

  for (let i = 0; i < cantidad; i++) {
    hoja.setRowHeight(filaInicio + i, 200);
  }

  hoja.setRowHeight(9, 200);

  hoja.getRange(filaInicio, 1, cantidad, columnas).setBorder(
    true, true, true, true,
    true, true,
    "black", SpreadsheetApp.BorderStyle.SOLID
  );

  const ultimaFila = filaInicio + cantidad - 1;
  const rangoUltima = hoja.getRange(ultimaFila, 1, 1, columnas);
  rangoUltima.breakApart();
  rangoUltima.merge();
  hoja.setRowHeight(ultimaFila, 20);
}

// ================= PDF COT =================

function generarPDFSoloZona() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaOriginal = ss.getSheetByName("FORMATO DE COT");

  const numeroCotizacion = hojaOriginal.getRange("F6").getValue();
  const nombreCliente = hojaOriginal.getRange("D3:F3").getValue();
  const cantidadFilas = parseInt(hojaOriginal.getRange("K2").getValue(), 10);

  if (!numeroCotizacion || !nombreCliente) {
    SpreadsheetApp.getUi().alert("Faltan datos.");
    return;
  }

  if (isNaN(cantidadFilas) || cantidadFilas <= 0) {
    SpreadsheetApp.getUi().alert("Cantidad inválida.");
    return;
  }

  const hojaCopia = hojaOriginal.copyTo(ss).setName("TEMP_PDF");

  hojaCopia.getDrawings().forEach(d => d.remove());

  const totalCols = hojaCopia.getMaxColumns();
  if (totalCols > 6) {
    hojaCopia.deleteColumns(7, totalCols - 6);
  }

  const data = hojaCopia.getDataRange().getValues();
  let ultimaFila = 1;

  for (let i = data.length - 1; i >= 0; i--) {
    if (data[i].join("").trim() !== "") {
      ultimaFila = i + 1;
      break;
    }
  }

  const totalFilas = hojaCopia.getMaxRows();
  if (ultimaFila < totalFilas) {
    hojaCopia.deleteRows(ultimaFila + 1, totalFilas - ultimaFila);
  }

  const nombrePDF = `Cotización_${numeroCotizacion}_${nombreCliente}`.replace(/[\\/:*?"<>|]/g, "") + ".pdf";

  const url = ss.getUrl().replace(/edit$/, '') +
    'export?exportFormat=pdf&format=pdf' +
    '&size=letter&portrait=true&fitw=true' +
    '&sheetnames=false&printtitle=false&pagenumbers=false&gridlines=false&fzr=false' +
    '&gid=' + hojaCopia.getSheetId();

  const blob = UrlFetchApp.fetch(url, {
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }
  }).getBlob().setName(nombrePDF);

  const carpeta = DriveApp.getFolderById('1CpkHInm6RbK8VDzmb2QH2UnST7JpqRya');
  const archivo = carpeta.createFile(blob);

  ss.deleteSheet(hojaCopia);

  SpreadsheetApp.getUi().alert("PDF generado:\n" + archivo.getName());
}

// ================= ON EDIT UNIFICADO =================

function onEdit(e) {
  const hoja = e.source.getActiveSheet();
  const fila = e.range.getRow();
  const columna = e.range.getColumn();

  if (fila === 1) return;

  const nombreHoja = hoja.getName();

  // ===== RESUMEN COT =====
  if (nombreHoja === "Resumen_Cot" && columna === 5) {
    const estado = e.range.getValue();
    const celdaFecha = hoja.getRange(fila, 6);
    const numeroCot = hoja.getRange(fila, 1).getValue();

    if (estado === "Aceptada" && !celdaFecha.getValue()) {
      const fecha = new Date();
      celdaFecha.setValue(fecha);
      celdaFecha.setNumberFormat("dd/MM/yyyy hh:mm:ss");

      const webhookURL = "https://script.google.com/macros/s/AKfycbxh8GYVq4zM2DeeigyXoPBb3A6TMRGvn1N7LhIjj63jnlzkUG8XwFVKKYvHQ5ChArlO/exec";

      UrlFetchApp.fetch(webhookURL, {
        method: "post",
        contentType: "application/json",
        payload: JSON.stringify({ numeroCot: numeroCot })
      });

    } else if (estado !== "Aceptada") {
      celdaFecha.setValue("");
    }
  }
}
